4//! DNF bitmap index queries over ordered bucket streams.
5//!
6//! Callers build a `BitmapQuery` as an OR of terms. Each term is an AND of
7//! signed dimension-key literals. Evaluation yields matching bitmap members as
8//! they are produced. Back-pressure from downstream consumers, e.g. a
9//! `.take(page_size)`, propagates back to the backend-provided bucket streams
10//! and avoids materializing matches we won't use.
11//!
12//! Queries are intentionally restricted to anchored DNF: every term must contain
13//! at least one positive literal. Positive literals give the evaluator concrete
14//! bitmap streams to scan and intersect; negative literals only shrink those
15//! candidate streams. Supporting negative-only terms such as `NOT sender = A`
16//! would require scanning an external universe for the requested range and
17//! subtracting from it, which defeats the index's selective streaming behavior
18//! and makes pagination depend on a full-range scan. Requiring DNF at the API
19//! boundary also keeps this evaluator as a set of ordered stream merge-joins
20//! instead of a recursive expression engine or a query normalizer.
21//!
22//! Backends provide one ordered `(bucket_id, RoaringBitmap)` stream or iterator
23//! per dimension key. The merge-join machinery here is storage-agnostic:
24//! BigTable, RocksDB, or any other backend can reuse it as long as its bucket
25//! source is sparse, ordered by the requested scan direction, and stores bitmap
26//! positions relative to that bucket.
27

54/// Terminal signal: the per-request bucket-fetch budget is exhausted.
55/// Surfaced as `anyhow::Error` so it short-circuits `try_stream!`
56/// pipelines through the existing error path. A silent EOF would be
57/// indistinguishable from a leaf reaching the range terminus, so the
58/// driver would advance the cursor to the end and claim full coverage
59/// instead of truncating the scan at the current floor.
60#[derive(Debug, thiserror::Error)]
61#[error("bitmap scan budget exhausted")]
62pub struct BitmapScanLimitExceeded;
63
64/// Aggregate of multiple terminal errors raised in the same driver round
65/// (e.g. two leaves both exhaust the shared budget before the next round).
66/// The single-error shortcut returns the inner error directly so existing
67/// downcasts on the wire keep working — `MultiError` only appears when 2+
68/// errors coincide, in which case downstream consumers should use
69/// [`error_contains`] to interrogate the aggregate.
70#[derive(Debug)]
71pub struct MultiError(Vec<anyhow::Error>);
72
73impl MultiError {
74    /// Wrap a non-empty error list into an `anyhow::Error`. With exactly
75    /// one error, returns it unwrapped so the common case preserves
76    /// `downcast_ref` behavior on the original concrete type.
77    pub fn collapse(mut errs: Vec<anyhow::Error>) -> anyhow::Error {
78        assert!(!errs.is_empty(), "MultiError::collapse on empty Vec");
79        if errs.len() == 1 {
80            return errs.pop().expect("len == 1");
81        }
82        anyhow::Error::new(MultiError(errs))
83    }
84
85    pub fn iter(&self) -> impl Iterator<Item = &anyhow::Error> {
86        self.0.iter()
87    }
88}
89
90impl std::fmt::Display for MultiError {
91    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
92        write!(f, "{} concurrent errors", self.0.len())?;
93        for (i, e) in self.0.iter().enumerate() {
94            write!(f, "\n  [{i}] {e}")?;
95        }
96        Ok(())
97    }
98}

102/// Downcast probe that looks through a `MultiError` aggregate. Returns
103/// the first inner `T` whether the top-level error is `T` directly or a
104/// `MultiError` containing one. Use this instead of `downcast_ref::<T>`
105/// at sites that may receive aggregated errors from the bitmap eval
106/// driver.
107pub fn error_contains<T: std::error::Error + Send + Sync + 'static>(
108    err: &anyhow::Error,
109) -> Option<&T> {
110    if let Some(t) = err.downcast_ref::<T>() {
111        return Some(t);
112    }
113    if let Some(multi) = err.downcast_ref::<MultiError>() {
114        return multi.iter().find_map(|e| e.downcast_ref::<T>());
115    }
116    None
117}

119/// Item or progress watermark flowing through a bitmap eval pipeline.
120/// `Watermark(p)` means every Item with position strictly before `p`
121/// in scan direction has been emitted upstream. Downstream stages must
122/// preserve watermark/item ordering — that's what makes the watermark a
123/// safe resume cursor on timeout.
124#[derive(Clone, Copy, Debug, Eq, PartialEq)]
125pub enum Watermarked<T> {
126    Item(T),
127    Watermark(u64),
128}
129
130impl<T> Watermarked<T> {
131    pub fn map_item<U>(self, f: impl FnOnce(T) -> U) -> Watermarked<U> {
132        match self {
133            Watermarked::Item(t) => Watermarked::Item(f(t)),
134            Watermarked::Watermark(p) => Watermarked::Watermark(p),
135        }
136    }
137}

223impl BitmapKey {
224    pub fn new(bytes: Vec<u8>) -> Result<Self> {

270impl BitmapQuery {
271    pub fn new(terms: Vec<BitmapTerm>) -> Result<Self> {
272        if terms.is_empty() {
273            bail!("bitmap query must contain at least one term");
274        }
275        Ok(Self { terms })
276    }
277
278    pub fn scan(dimension_key: Vec<u8>) -> Result<Self> {
279        Ok(Self {
280            terms: vec![BitmapTerm::new(vec![BitmapLiteral::include(
281                dimension_key,
282            )?])?],
283        })
284    }
285
286    /// Count of distinct dimension-key leaves the query will scan. Identical
287    /// keys across literals (whether within a term or across terms) collapse
288    /// to one leaf at evaluation time, so the per-request budget floor —
289    /// "every leaf can emit its first watermark" — applies to this
290    /// deduplicated count, not the raw literal total.
291    pub fn unique_leaf_count(&self) -> usize {
292        self.terms
293            .iter()
294            .flat_map(|t| t.literals.iter().map(|l| l.key_bytes()))
295            .collect::<HashSet<_>>()
296            .len()
297    }
298}

300impl BitmapTerm {
301    pub fn new(literals: Vec<BitmapLiteral>) -> Result<Self> {
302        if !literals
303            .iter()
304            .any(|literal| matches!(literal, BitmapLiteral::Include(_)))
305        {
306            bail!("bitmap query term must contain at least one include literal");
307        }
308        Ok(Self { literals })
309    }

312/// Deduplicated leaf list + per-term references over those leaves, ready for
313/// the evaluator. `keys[i]` is the dimension-key bytes for leaf `i`; each
314/// [`TermSpec`] references leaves by that index.
315pub(crate) struct DedupedQuery {
316    pub(crate) keys: Vec<Vec<u8>>,
317    pub(crate) terms: Vec<TermSpec>,
318}
319
320/// Deduplicate literals across the whole query and translate each term's
321/// includes/excludes into indices over the shared leaf list.
322///
323/// Dedup is keyed on encoded dimension-key bytes, so the same key appearing
324/// in multiple terms (e.g. `(sender=A AND module=X) OR (sender=A AND
325/// type=Y)`) maps to a single backend scan instead of two duplicate scans.
326pub(crate) fn build_term_specs(terms: Vec<BitmapTerm>) -> DedupedQuery {
327    let mut key_to_idx: HashMap<Vec<u8>, usize> = HashMap::new();
328    let mut keys: Vec<Vec<u8>> = Vec::new();
329    let mut specs: Vec<TermSpec> = Vec::with_capacity(terms.len());
330    for term in terms {
331        let mut include_idx = Vec::with_capacity(term.literals.len());
332        let mut exclude_idx = Vec::with_capacity(term.literals.len());
333        for literal in term.literals {
334            let (push_target, key) = match literal {
335                BitmapLiteral::Include(k) => (&mut include_idx, k.into_inner()),
336                BitmapLiteral::Exclude(k) => (&mut exclude_idx, k.into_inner()),
337            };
338            let idx = match key_to_idx.get(&key) {
339                Some(&i) => i,
340                None => {
341                    let i = keys.len();
342                    key_to_idx.insert(key.clone(), i);
343                    keys.push(key);
344                    i
345                }
346            };
347            push_target.push(idx);
348        }
349        specs.push(TermSpec {
350            includes: include_idx,
351            excludes: exclude_idx,
352            unsatisfiable: false,
353        });
354    }
355    DedupedQuery { keys, terms: specs }
356}

358/// The less-advanced of two frontier positions in scan direction: the min
359/// ascending, the max descending. Used to keep a merged frontier bounded by
360/// the slowest contributor.
361fn bound_in_direction(a: u64, b: u64, direction: ScanDirection) -> u64 {
362    match direction {
363        ScanDirection::Ascending => a.min(b),
364        ScanDirection::Descending => a.max(b),
365    }
366}
367
368/// Whether emitting `next` as a watermark advances the frontier past the
369/// previously emitted one. Ascending frontiers strictly increase,
370/// descending strictly decrease; the first watermark always advances.
371fn frontier_advanced(prev: Option<u64>, next: u64, direction: ScanDirection) -> bool {
372    match prev {
373        None => true,
374        Some(prev) => match direction {
375            ScanDirection::Ascending => next > prev,
376            ScanDirection::Descending => next < prev,
377        },
378    }
379}

381/// Clamped member-id edges of `bucket` in scan direction: `(pre, post)` where
382/// `pre` is the leading edge (everything before it is already covered) and
383/// `post` is the trailing edge (everything up to and including the bucket is
384/// covered). Ascending: `(low, high)`; descending: `(high, low)`. Both clamped
385/// to the request range so cursors stay in-bounds when they round-trip into a
386/// follow-up request with a different range.
387pub(crate) fn bucket_edges(
388    bucket: u64,
389    bucket_size: u64,
390    range: &Range<u64>,
391    direction: ScanDirection,
392) -> (u64, u64) {
393    let start = bucket.saturating_mul(bucket_size);
394    let end = start.saturating_add(bucket_size);
395    match direction {
396        ScanDirection::Ascending => (start.max(range.start), end.min(range.end)),
397        ScanDirection::Descending => (end.min(range.end), start.max(range.start)),
398    }
399}

401/// Pull the snapshotted bitmap for leaf `i` to give to one referencing term
402/// this round. Returns `None` if the leaf isn't on the floor bucket (so the
403/// term short-circuits). When `remaining_refs[i] > 1`, the bitmap is cloned so
404/// other referencing terms still see it; the last reference takes by value to
405/// avoid an unnecessary copy in the common single-reference case.
406pub(crate) fn take_snapshot_bitmap(
407    snapshot: &mut [Option<RoaringBitmap>],
408    remaining_refs: &mut [usize],
409    on_floor: &[bool],
410    i: usize,
411) -> Option<RoaringBitmap> {
412    if !on_floor[i] {
413        return None;
414    }
415    if remaining_refs[i] > 1 {
416        remaining_refs[i] -= 1;
417        snapshot[i].clone()
418    } else {
419        remaining_refs[i] = remaining_refs[i].saturating_sub(1);
420        snapshot[i].take()
421    }
422}
423
424/// Per-round refcount: how many satisfiable term-side slots reference each
425/// on-floor leaf. Drives [`take_snapshot_bitmap`]'s take-vs-clone decision so
426/// the last referencing slot reclaims the bitmap by value.
427pub(crate) fn count_on_floor_refs(terms: &[TermSpec], on_floor: &[bool]) -> Vec<usize> {
428    let mut refs = vec![0usize; on_floor.len()];
429    for term in terms {
430        if term.unsatisfiable {
431            continue;
432        }
433        for &i in term.includes.iter().chain(term.excludes.iter()) {
434            if on_floor[i] {
435                refs[i] += 1;
436            }
437        }
438    }
439    refs
440}

442/// Recompute leaf liveness from current term state. A leaf becomes
443/// `unreferenced` when no satisfiable term still points at it, or when its
444/// head is at EOF (the bucket stream is permanently exhausted; any further
445/// peek would be wasted work, and any include-referencing term will be marked
446/// `unsatisfiable` separately).
447///
448/// `unreferenced` is monotonic — entries only transition false → true — so
449/// this is safe to invoke each round; previously-retired leaves stay retired.
450pub(crate) fn recompute_unreferenced(
451    terms: &[TermSpec],
452    class: &[Option<LeafHead>],
453    unreferenced: &mut [bool],
454) {
455    let leaf_count = unreferenced.len();
456    let mut referenced = vec![false; leaf_count];
457    for term in terms {
458        if term.unsatisfiable {
459            continue;
460        }
461        for &i in term.includes.iter().chain(term.excludes.iter()) {
462            if !unreferenced[i] && !matches!(class[i], Some(LeafHead::Eof)) {
463                referenced[i] = true;
464            }
465        }
466    }
467    for i in 0..leaf_count {
468        if !referenced[i] {
469            unreferenced[i] = true;
470        }
471    }
472}

474/// Evaluate one DNF term at a single bucket from the per-leaf bitmaps present
475/// there: intersect the includes (any absent include ⇒ empty term), then
476/// subtract the union of the present excludes (`a AND NOT b`). Returns the
477/// term's matches at the bucket, or `None` if empty. Bitmaps are taken by value
478/// so the caller hands over the consumed leaf rows without cloning.
479pub(crate) fn eval_term_at_bucket(
480    includes: Vec<Option<RoaringBitmap>>,
481    excludes: Vec<Option<RoaringBitmap>>,
482) -> Option<RoaringBitmap> {
483    let mut acc: Option<RoaringBitmap> = None;
484    for include in includes {
485        // A missing include means the intersection is empty at this bucket.
486        let bitmap = include?;
487        acc = Some(match acc {
488            None => bitmap,
489            Some(a) => a & bitmap,
490        });
491    }
492    // Anchored terms always carry at least one include, so `acc` is `Some`.
493    let mut acc = acc?;
494    for exclude in excludes.into_iter().flatten() {
495        acc -= exclude;
496    }
497    (!acc.is_empty()).then_some(acc)
498}

508}
